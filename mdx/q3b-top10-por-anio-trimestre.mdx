-- Pregunta 3 (distribución en el tiempo): ventas de los 10 productos más rentables por trimestre.
SELECT
  NON EMPTY [Fecha].[Año-Trimestre-Mes].Levels(2).MEMBERS ON COLUMNS,
  TOPCOUNT( NONEMPTY( [Producto].[Categoría-Subcategoría-Producto].Levels(3).MEMBERS, [Measures].[Utilidad Total] ),
            10, [Measures].[Utilidad Total] ) ON ROWS
FROM [Ventas AdventureWorks]
WHERE ( [Measures].[Total Ventas] )
